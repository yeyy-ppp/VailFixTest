}

    private static Option option = new Option();

    public static OptionBuilder withLongOpt(String longOpt) {
        option.longOpt = longOpt;
        return this;
    }

    public static OptionBuilder withOpt(String opt) {
        option.opt = opt;
        return this;
    }

    public static OptionBuilder hasArg() {
        option.args = 1;
        return this;
    }

    public static OptionBuilder hasArg(boolean hasArg) {
        option.args = hasArg ? 1 : UNINITIALIZED;
        return this;
    }

    public static OptionBuilder hasOptionalArg() {
        option.optionalArg = true;
        option.args = 1;
        return this;
    }

    public static void reset() {
        option = new Option();
    }

    public static Option create() {
        Option result = option;
        reset();
        return result;
    }
}
