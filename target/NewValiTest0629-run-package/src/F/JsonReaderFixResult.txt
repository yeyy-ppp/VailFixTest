        int stackSize = stackSizeField.getInt(reader);

        return stack[stackSize - 1];
    }

    private boolean isScope(JsonReader reader, int scope) {
        try {
            return getScope(reader) == scope;
        } catch (Exception e) {
            return false;
        }
    }
}
